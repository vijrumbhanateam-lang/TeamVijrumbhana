function doPost(e) { try { const data = JSON.parse(e.postData.contents); const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); // MEMBERSHIP REGISTRATION if (data.reason !== undefined && data.skills !== undefined) { let sheet = spreadsheet.getSheetByName("Membership"); if (!sheet) { sheet = spreadsheet.insertSheet("Membership"); } // Add headings if the sheet is empty if (sheet.getLastRow() === 0) { sheet.appendRow([ "Enrollment No", "Date", "Student Name", "Class", "Section", "Roll Number", "Mobile", "Email", "Reason", "Skills", "Status" ]); } const enrollmentNo = "TVM-" + Utilities.getUuid().substring(0, 8).toUpperCase(); sheet.appendRow([ enrollmentNo, new Date(), data.studentName || "", data.className || "", data.section || "", data.rollNumber || "", data.mobile || "", data.email || "", data.reason || "", data.skills || "", "Pending" ]); return ContentService .createTextOutput(JSON.stringify({ success: true, message: "Membership application saved successfully" })) .setMimeType(ContentService.MimeType.JSON); } // COMPLAINT REGISTRATION let sheet = spreadsheet.getActiveSheet(); if (sheet.getName() === "Membership") { sheet = spreadsheet.getSheetByName("Complaints") || spreadsheet.getSheetByName("Complaint") || spreadsheet.getSheets()[0]; } sheet.appendRow([ new Date(), data.studentName || "", data.class || "", data.rollNumber || "", data.section || "", data.phoneNumber || "", data.complaintDetails || "", data.institutionType || "", data.institutionName || "" ]); return ContentService .createTextOutput(JSON.stringify({ success: true, message: "Complaint saved successfully" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ success: false, message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { return ContentService.createTextOutput( "Team Vijrumbhana connection is working!" ); }
